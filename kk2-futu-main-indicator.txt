{KK2富途牛牛主图指标}
{参数：SD=20，最大120，最小2}
{参数：WIDTH=0，最大100，最小0}
{参数：N=15，最大1000，最小0}

{==================== 布林线 ====================}
DIS:=STDP(CLOSE,SD);
MID:MA(CLOSE,SD),COLORFFAEC9;
UPPER:MID+WIDTH*DIS,COLORFFC90E;
LOWER:MID-WIDTH*DIS,COLOR0CAEE6;

{==================== 获利筹复合指标 ====================}
VAR1:=(CLOSE-LLV(LOW,13))/(HHV(HIGH,13)-LLV(LOW,13))*100;
VAR2:=(CLOSE-LLV(LOW,9))/(HHV(HIGH,9)-LLV(LOW,9))*100;
VAR3:=(SMA(VAR2,3,1)-18)*1.55;
VAR4:=SMA(VAR3,3,1);
VAR5:=(CLOSE-LLV(LOW,100))/(HHV(HIGH,100)-LLV(LOW,100))*100;
VAR6:=SMA(VAR5,10,1);
VAR7:=(EMA(SMA(VAR6,8,1),34)-25)*2.6;
VAR8:=POW(MA(CLOSE,5),2)+MA(CLOSE,5);
VAR9:=POW(MA(LOW,5),2)+MA(LOW,5);
VARA:=POW(MA(HIGH,5),2)+MA(HIGH,5);
VARB:=((VAR8-LLV(VAR9,64))/(HHV(VARA,64)-LLV(VAR9,64))*150+65-10)/2;
VARC:=SMA(VARB,3,1)*1.5-46;
VARD:=SMA(VARC,3,1);
VARE:=3*VARC-2*VARD;
VARF:=(VAR1+VAR2+VAR5)/3;
VAR10:=((VAR3+VAR3+VAR6+VARC)/4-15)*1.67;
VAR11:=((VAR4+VAR4+VAR7+VARD)/4-15)*1.67;
VAR12:=((VAR1+VAR5*2+VARF+VAR10+VAR11+VARB+VARC+VARD+VARE)/10-15)*1.67;
VAR13:=(HHV(HIGH,5)-CLOSE)/(HHV(HIGH,5)-LLV(LOW,5))*(-1)+0.9;
VAR14:=(HHV(HIGH,10)-CLOSE)/(HHV(HIGH,10)-LLV(LOW,10))*(-1)+0.92;
VAR15:=(HHV(HIGH,15)-CLOSE)/(HHV(HIGH,15)-LLV(LOW,15))*(-1)+0.93;
VAR16:=(HHV(HIGH,55)-CLOSE)/(HHV(HIGH,55)-LLV(LOW,55))*(-1)+0.94;
VAR17:=(HHV(HIGH,89)-CLOSE)/(HHV(HIGH,89)-LLV(LOW,89))*(-1)+0.95;
VAR18:=(HHV(HIGH,120)-CLOSE)/(HHV(HIGH,120)-LLV(LOW,120))*(-1)+0.91;
VAR19:=(HHV(HIGH,180)-CLOSE)/(HHV(HIGH,180)-LLV(LOW,180))*(-1)+0.96;
VAR1A:=((VAR13*8+VAR14*8+VAR15*8+VAR16+VAR17+VAR18+VAR19)/28-0.1)*185;
VAR1B:=SMA(VAR1A,3,1);
VAR1C:=(SMA(VAR1B,8,1)-8)*1.18;
获利筹:=(VAR3+VAR6+VAR1B*2)/4;
V2:=获利筹;
V3:=SMA(获利筹,3,1);

{==================== MACD ====================}
SHORT:=12;
LONG:=26;
M:=9;
DIF:=EMA(CLOSE,SHORT)-EMA(CLOSE,LONG);
DEA:=EMA(DIF,M);
MACD:=(DIF-DEA)*2;

{==================== RSI ====================}
P1:=6;
P2:=12;
P3:=24;
LC:=REF(CLOSE,1);
TEMP1:=MAX(CLOSE-LC,0);
TEMP2:=ABS(CLOSE-LC);
RSI1:=SMA(TEMP1,P1,1)/SMA(TEMP2,P1,1)*100;
RSI2:=SMA(TEMP1,P2,1)/SMA(TEMP2,P2,1)*100;
RSI3:=SMA(TEMP1,P3,1)/SMA(TEMP2,P3,1)*100;

{==================== 成交量判断 ====================}
VOL1:=MA(VOLA,1);
VOL2:=MA(REF(VOLA,1),3);
VOL3:=MA(VOLA,2);
VOL4:=MA(REF(VOLA,2),3);
VOL5:=MA(VOLA,3);
VOL6:=MA(REF(VOLA,3),3);
VOLC_B:=VOL1>VOL2*1.60 OR VOL3>VOL4*1.60 OR VOL5>VOL6*1.60;
VOLC_B_:=VOL1>VOL2*1.80 OR VOL3>VOL4*1.80 OR VOL5>VOL6*1.80;
VOLC_S:=VOL1>VOL2*1.10 OR VOL3>VOL4*1.10 OR VOL5>VOL6*1.10;

{==================== 绝反信号 ====================}
VARA1:=(CLOSE-LLV(LOW,60))/(HHV(HIGH,60)-LLV(LOW,60))*80;
VARB1:=SMA(VARA1,7,1);
VARC1:=SMA(VARB1,5,1);
VAR2N:=LLV(LOW,3)<=LLV(LOW,60);
VAR3N:=C>OPEN AND (C/OPEN>1.03 OR C>1.03*REF(C,1));
绝反:=IF(VAR2N AND VAR3N AND VOLC_B,20,0);

{==================== 买入评分 ====================}
B_SCORE_1:=IF(REF(获利筹,1)<8 AND 获利筹>REF(获利筹,1),1,0);
B_SCORE_2:=IF(获利筹>30 AND 获利筹<80 AND C>1.02*REF(C,1) AND VOLA>MA(VOLA,10)*2.0 AND VOLA<MA(VOLA,100)*5.0,1,0);
B_SCORE_3:=IF(CROSS(V2,V3),1,0)+IF(V2>REF(V2,1),1,0);
B_SCORE_4:=IF(REF(MACD,1)<REF(MACD,2) AND MACD>REF(MACD,1),1,0)+IF(REF(CLOSE,1)<REF(LOWER,1) AND CLOSE>0.2*MID+0.8*LOWER,1,0);
B_SCORE_5:=IF(VOLC_B,1,0);
B_SCORE:=B_SCORE_1+B_SCORE_2+B_SCORE_3+B_SCORE_4+B_SCORE_5;
BS:=B_SCORE;
B_CONDITION:=(B_SCORE>=5 AND (B_SCORE_1 OR B_SCORE_2)) OR (B_SCORE>=4 AND (B_SCORE_1 OR B_SCORE_2) AND (B_SCORE_5 OR REF(获利筹,1)<-5));
B_SCORE_V:=IF(绝反,10,0);

{==================== 卖出评分 ====================}
S_SCORE_1:=IF(HHV(获利筹,3)>0.96*HHV(REF(获利筹,20),150),1,0)+IF(HHV(获利筹,2)>85,1,0)+IF(HHV(获利筹,2)>100,1,0);
S_SCORE_2:=IF(CROSS(V3,V2),1,0)+IF(V2<REF(V2,1),1,0)+IF(REF(MACD,1)>REF(MACD,2) AND MACD<REF(MACD,1),1,0);
S_SCORE_3:=IF(LLV(REF(MACD,1),15)>0 AND MACD<0,1,0)+IF(REF(CLOSE,1)>REF(UPPER,1) AND CLOSE>0.2*MID+0.8*UPPER,1,0)+IF(VOLC_S,1,0);
S_SCORE_4:=IF(CROSS(RSI1,85),1,0);
S_SCORE:=S_SCORE_1+S_SCORE_2+S_SCORE_3+S_SCORE_4;
S_CONDITION:=S_SCORE>=7 OR (S_SCORE>=6 AND SMA(S_SCORE,3,1)>=5.0);
DOWN_LONG:=获利筹>70 AND MA(获利筹,100)>15 AND MA(获利筹,100)<50;
S_CONDITION_DOWN_LONG:=DOWN_LONG AND S_SCORE>=5;
UP_LAST:=MA(获利筹,100)>65 AND HHV(获利筹,3)<HHV(获利筹,30)*0.9;
S_CONDITION_UP_LAST:=UP_LAST AND S_SCORE>=5;
SS:=S_SCORE;
H90_:=90;
B_POSITION:=0.995*LOW;
S_POSITION:=1.008*HIGH;

{==================== 神奇9转 ====================}
A1:=C>REF(C,4);
NT:=BARSLASTCOUNT(A1);
TJ11:=NT=9;
TJ13:=ISLASTBAR AND BETWEEN(NT,5,8);
AY:=(BACKSET(TJ11>0,9) OR BACKSET(TJ13>0,NT))*NT;
DRAWNUMBER(AY>0 AND AY<9,H,AY,0,N),COLORFF00FF;
DRAWNUMBER(NT=9,H,9,0,N),COLORGREEN;

B1:=C<REF(C,4);
NT0:=BARSLASTCOUNT(B1);
TJ21:=NT0=9;
TJ23:=ISLASTBAR AND BETWEEN(NT0,5,8);
AY1:=(BACKSET(TJ21>0,9) OR BACKSET(TJ23>0,NT0))*NT0;
DRAWNUMBER(AY1>0 AND AY1<9,L,AY1,0,-1*N),COLORGREEN;
DRAWNUMBER(NT0=9,L,9,0,-1*N),COLORFF00FF;

{==================== 主图信号输出 ====================}
DRAWICON(绝反,LOW,34);
DRAWICON(B_CONDITION,LOW,7);
DRAWICON(S_CONDITION_UP_LAST,HIGH,8);
